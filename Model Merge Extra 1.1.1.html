# Description: A ComfyUI custom node to adaptively convert or hybrid-blend the prediction type of a diffusion model.

import torch
import copy
import comfy.utils

class AdaptiveModelWrapper(torch.nn.Module):
    """
    An advanced functional wrapper for a UNet model.
    It intercepts the forward pass to apply calibration and then performs
    a principled blend between the model's original epsilon prediction
    and its mathematically converted v-prediction.
    """
    def __init__(self, wrapped_model, model_sampling, source_type, target_type, correction_factor=1.0, hybrid_alpha=1.0):
        super().__init__()
        self.wrapped_model = wrapped_model
        self.model_sampling = model_sampling # This is needed to get the sigmas
        self.source_type = source_type
        self.target_type = target_type
        self.correction_factor = correction_factor
        self.hybrid_alpha = hybrid_alpha

    def get_x0(self, pred, x, sigma, pred_type):
        """Calculates the predicted x0 (original image) from a given prediction."""
        alpha_t = 1.0 / (sigma ** 2 + 1.0).sqrt()
        sigma_t = sigma

        if pred_type == 'epsilon':
            # Clamping to avoid potential numerical instability with fp16
            if alpha_t.dtype == torch.float16:
                alpha_t = torch.max(alpha_t, torch.tensor(0.001, dtype=torch.float16, device=alpha_t.device))
            return (x - sigma_t * pred) / alpha_t
        elif pred_type == 'v_prediction':
            return alpha_t * x - sigma_t * pred
        else:
            raise ValueError(f"Unknown prediction type: {pred_type}")

    def to_target_prediction(self, x0, x, sigma):
        """Converts a predicted x0 back to the target prediction type for the sampler."""
        alpha_t = 1.0 / (sigma ** 2 + 1.0).sqrt()
        sigma_t = sigma
        
        # Clamping to avoid division by zero at the last step
        if sigma_t.dtype == torch.float16:
            sigma_t = torch.max(sigma_t, torch.tensor(0.001, dtype=torch.float16, device=sigma_t.device))

        if self.target_type == 'epsilon':
            return (x - alpha_t * x0) / sigma_t
        elif self.target_type == 'v_prediction':
            # First, derive epsilon from x0, then convert epsilon to v-prediction
            eps = (x - alpha_t * x0) / sigma_t
            return alpha_t * eps - sigma_t * x0
        else:
            # Should not happen with current options, but as a fallback
            return x0

    def forward(self, x, timesteps, context, **kwargs):
        # 1. Get the model's native prediction by calling the wrapped model.
        original_pred = self.wrapped_model(x, timesteps, context=context, **kwargs)
        
        # 2. Get the sigma for the current timestep from the stored model_sampling object.
        if not hasattr(self.model_sampling, 'sigmas'):
            raise AttributeError("The provided 'model_sampling' object must have a 'sigmas' attribute.")
        sigmas = self.model_sampling.sigmas
        sigma = sigmas[timesteps.long()].view(-1, 1, 1, 1)

        # 3. Apply calibration factor to the source prediction (if it's epsilon).
        calibrated_pred = original_pred
        if self.source_type == 'epsilon':
            calibrated_pred = original_pred * self.correction_factor

        # If no blending is needed (alpha=0), just perform a direct mathematical conversion and return.
        if self.hybrid_alpha <= 0.0:
            return self.convert_prediction(calibrated_pred, x, sigma)

        # --- Principled Blending via x0 Interpolation ---
        # 4. Calculate the predicted x0 from the model's native (calibrated) prediction.
        x0_from_source = self.get_x0(calibrated_pred, x, sigma, self.source_type)

        # 5. Calculate the predicted x0 from the mathematically converted prediction.
        converted_pred = self.convert_prediction(calibrated_pred, x, sigma)
        x0_from_converted = self.get_x0(converted_pred, x, sigma, self.target_type)
        
        # 6. Linearly interpolate between the two x0 predictions. This is the "principled blend".
        blended_x0 = (1.0 - self.hybrid_alpha) * x0_from_source + self.hybrid_alpha * x0_from_converted
        
        # 7. Convert the final, blended x0 back into the target prediction format that the sampler expects.
        final_prediction = self.to_target_prediction(blended_x0, x, sigma)
        
        return final_prediction

    def convert_prediction(self, pred, x, sigma):
        """Helper function for direct mathematical conversion when no blending is needed."""
        if self.source_type == self.target_type:
            return pred
        
        alpha_t = 1.0 / (sigma ** 2 + 1.0).sqrt()
        sigma_t = sigma

        if self.source_type == 'epsilon' and self.target_type == 'v_prediction':
            return alpha_t * pred - sigma_t * x
        elif self.source_type == 'v_prediction' and self.target_type == 'epsilon':
            # Clamp alpha_t to avoid division by zero, especially at high sigmas
            if alpha_t.dtype == torch.float16:
                alpha_t = torch.max(alpha_t, torch.tensor(0.001, dtype=torch.float16, device=alpha_t.device))
            return (pred + sigma_t * x) / alpha_t
        else:
            return pred

    def __getattr__(self, name):
        try:
            return super().__getattr__(name)
        except AttributeError:
            # Fallback to getting attributes from the wrapped model for compatibility
            return getattr(self.wrapped_model, name)


class AdaptivePredictionConverter:
    @classmethod
    def INPUT_TYPES(s):
        return {
            "required": {
                "model": ("MODEL",),
                "target_prediction": (["epsilon", "v_prediction"],),
                "calibration_steps": ("INT", {"default": 15, "min": 1, "max": 100, "step": 1}),
                "calibration_strength": ("FLOAT", {"default": 1.0, "min": 0.0, "max": 2.0, "step": 0.05}),
                "hybrid_alpha": ("FLOAT", {"default": 0.65, "min": 0.0, "max": 1.0, "step": 0.01}),
            }
        }

    RETURN_TYPES = ("MODEL",)
    FUNCTION = "convert_and_calibrate"
    CATEGORY = "model_patches/experimental"

    def calibrate_model(self, model, num_steps):
        print(f"Starting model calibration for {num_steps} steps...")
        
        try:
            unet = model.model.diffusion_model
            model_sampling = model.model.model_sampling
        except AttributeError as e:
            print(f"Warning: Could not find required attributes for calibration: {e}. Cannot calibrate.")
            return 1.0

        device = model.load_device
        
        if not hasattr(unet, 'model_config') or not hasattr(unet.model_config, 'unet_config'):
            print("Warning: Model configuration not found. Cannot determine context_dim for calibration.")
            return 1.0
        context_dim = unet.model_config.unet_config.get('context_dim', 768)

        dummy_latent = torch.zeros((1, 4, 64, 64), device=device)
        dummy_context = torch.zeros((1, 77, context_dim), device=device)
        
        sigmas = model_sampling.sigmas
        if sigmas is None:
            print("Warning: Model does not have a sigma schedule. Cannot calibrate.")
            return 1.0

        measured_stds = []
        pbar = comfy.utils.ProgressBar(num_steps)
        
        for i in range(num_steps):
            sigma_index = max(0, min(len(sigmas) - 1, (len(sigmas) - 1) // 2 + i - num_steps // 2))
            sigma = sigmas[sigma_index].to(device)
            noise = torch.randn_like(dummy_latent) * sigma
            timestep = torch.tensor([sigma_index], device=device)

            with torch.no_grad():
                prediction = unet(noise, timestep, context=dummy_context).cpu()
            
            measured_stds.append(prediction.std().item())
            pbar.update(1)

        avg_std = sum(measured_stds) / len(measured_stds) if measured_stds else 1.0
        correction_factor = 1.0 / avg_std if avg_std != 0 else 1.0
        
        print(f"Calibration complete. Average STD: {avg_std:.4f}, Calculated Correction Factor: {correction_factor:.4f}")
        return correction_factor

    def convert_and_calibrate(self, model, target_prediction, calibration_steps, calibration_strength, hybrid_alpha):
        try:
            source_prediction = model.model.model_config.prediction_type
        except AttributeError:
            print("Warning: Source prediction_type not found. Assuming 'epsilon'.")
            source_prediction = "epsilon"

        correction_factor = 1.0
        if source_prediction == 'epsilon':
            correction_factor = self.calibrate_model(model, calibration_steps)
        
        final_factor = 1.0 + (correction_factor - 1.0) * calibration_strength
        if final_factor != 1.0:
            print(f"Applying final correction factor of {final_factor:.4f} (Strength: {calibration_strength * 100}%)")

        model_copy = model.clone()
        
        try:
            unet_to_wrap = model_copy.model.diffusion_model
            model_sampling = model_copy.model.model_sampling
        except AttributeError as e:
            print(f"ERROR: Could not find required attributes ('diffusion_model' or 'model_sampling') in the model structure: {e}. Aborting patch.")
            return (model,) # Return original model on failure

        wrapper = AdaptiveModelWrapper(unet_to_wrap, model_sampling, source_prediction, target_prediction, final_factor, hybrid_alpha)
        
        model_copy.model.diffusion_model = wrapper
        
        # Update the model's metadata so the sampler knows what to expect
        if hasattr(model_copy, 'model_options'):
            model_copy.model_options['prediction_type'] = target_prediction
        else:
            model_copy.model_options = {'prediction_type': target_prediction}

        return (model_copy,)
